Show/Hide Toggle Button Version 1
import React, { useState } from "react";
const App = () => {
  const [show, setShow] = useState(false);
  const buttonClicked = () => {
    if (show === true) {
      setShow(false);
    } else {
      setShow(true);
    }
  };
  return (
    <div>
      <button onClick={buttonClicked}>{show ? "Hide" : "Show"}</button>
      <div>{show && "This is the Secret Message"}</div>
    </div>
  );
};
export default App;
