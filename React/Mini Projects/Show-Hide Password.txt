// Show/Hide Password
import React, { useState } from "react";
const App = () => {
  const [isPassword, setIsPassword] = useState(false);
  return (
    <div>
      <div>
        <input
          type={isPassword ? "text" : "password"}
          placeholder="Enter the Password"
        />
        <button onClick={() => setIsPassword(!isPassword)}>
          {isPassword ? "Hide" : "Show"}
        </button>
      </div>
    </div>
  );
};
export default App;
