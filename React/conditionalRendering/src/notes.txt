Show/Hide Toggle Button Version 1
import React, { useState } from "react";
const App = () => {
  const [show, setShow] = useState(false);
  const buttonClicked = () => {
    if (show === true) {
      setShow(false);
    } else {
      setShow(true);
    }
  };
  return (
    <div>
      <button onClick={buttonClicked}>{show ? "Hide" : "Show"}</button>
      <div>{show && "This is the Secret Message"}</div>
    </div>
  );
};
export default App;

// Show/Hide Password
import React, { useState } from "react";
const App = () => {
  const [isPassword, setIsPassword] = useState(false);
  return (
    <div>
      <div>
        <input
          type={isPassword ? "text" : "password"}
          placeholder="Enter the Password"
        />
        <button onClick={() => setIsPassword(!isPassword)}>
          {isPassword ? "Hide" : "Show"}
        </button>
      </div>
    </div>
  );
};
export default App;
